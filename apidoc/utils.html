📝 로깅 시스템

Atio의 로깅 시스템은 라이브러리의 동작을 추적하고 디버깅을 위한 상세한 정보를 제공합니다.

🔍 setup_logger()

구조화된 로깅을 위한 로거를 설정합니다.

💡 로깅의 필요성

대용량 데이터 처리 시 다음과 같은 정보가 필요합니다:

  • 처리 진행 상황: 현재 처리 중인 단계와 완료율
  • 성능 메트릭: 처리 시간, 메모리 사용량, 처리 속도
  • 오류 추적: 실패 지점과 원인 분석
  • 디버깅 정보: 내부 상태와 중간 결과
# 로깅 시스템 설정 예시
import atio
import logging

# 로거 설정
logger = atio.setup_logger('my_app', level=logging.INFO)

# 로깅 사용 예시
def process_large_dataset():
    logger.info("대용량 데이터셋 처리 시작")
    
    # 데이터 로드
    df = pd.read_csv('large_dataset.csv')
    logger.info(f"데이터 로드 완료: {len(df):,}행, {len(df.columns)}열")
    
    # 처리 시작
    start_time = time.time()
    atio.write(df, 'processed.parquet', format='parquet', verbose=True)
    
    # 처리 완료
    elapsed_time = time.time() - start_time
    logger.info(f"처리 완료: {elapsed_time:.2f}초 소요")
    
    return True

📊 ProgressBar

데이터 처리 진행률을 시각적으로 표시하는 진행률 바입니다.

💡 진행률 표시의 필요성

대용량 데이터 처리 시 사용자는 다음과 같은 정보를 필요로 합니다:

  • 현재 진행률: 전체 작업 중 완료된 비율
  • 예상 완료 시간: 남은 시간과 완료 예정 시각
  • 처리 속도: 초당 처리량과 데이터 크기
  • 현재 상태: 어떤 작업을 수행 중인지
# 진행률 바 사용 예시
import atio
import time

def demonstrate_progress_bar():
    """진행률 바 데모"""
    # 진행률 바 생성
    progress = atio.ProgressBar(total=100, desc="데이터 처리")
    
    # 시뮬레이션된 작업
    for i in range(100):
        # 작업 시뮬레이션
        time.sleep(0.05)  # 50ms 대기
        
        # 진행률 업데이트
        progress.update(1)
        
        # 중간 상태 출력
        if i % 20 == 0:
            print(f"\\n현재 단계: {i}% 완료")
            print(f"처리된 항목: {i:,}개")
    
    progress.close()
    print("\\n✅ 모든 작업 완료!")

# 실제 데이터 처리에서의 진행률 바 사용
def process_with_progress():
    """실제 데이터 처리에서 진행률 표시"""
    import pandas as pd
    
    # 대용량 데이터 생성
    data = []
    total_rows = 10000
    
    # 진행률 바 설정
    progress = atio.ProgressBar(total=total_rows, desc="데이터 생성")
    
    for i in range(total_rows):
        # 데이터 생성
        row = {
            'id': i,
            'value': np.random.randn(),
            'category': np.random.choice(['A', 'B', 'C'])
        }
        data.append(row)
        
        # 진행률 업데이트
        progress.update(1)
    
    progress.close()
    
    # DataFrame 생성
    df = pd.DataFrame(data)
    
    # 저장 시에도 진행률 표시
    atio.write(df, 'generated_data.parquet', 
              format='parquet', 
              show_progress=True)
    
    print(f"\\n✅ {len(df):,}행 데이터 생성 및 저장 완료")

# JSON 읽기/쓰기 예시
import atio
import json

def json_processing_example():
    """JSON 처리 예시"""
    
    # 1. JSON 데이터 생성
    data = {
        'dataset_info': {
            'name': 'Customer Data',
            'version': '1.0',
            'created_at': '2024-01-15T10:30:00Z',
            'rows': 10000,
            'columns': ['id', 'name', 'email', 'age']
        },
        'processing_config': {
            'chunk_size': 1000,
            'compression': 'gzip',
            'format': 'parquet'
        },
        'metadata': {
            'author': 'Data Team',
            'description': 'Customer information dataset',
            'tags': ['customer', 'demographics', 'marketing']
        }
    }
    
    # 2. JSON 파일로 저장
    atio.write_json(data, 'config.json', indent=2)
    print("✅ JSON 파일 저장 완료")
    
    # 3. JSON 파일 읽기
    loaded_data = atio.read_json('config.json')
    print("✅ JSON 파일 읽기 완료")
    
    # 4. 데이터 확인
    print(f"데이터셋: {loaded_data['dataset_info']['name']}")
    print(f"버전: {loaded_data['dataset_info']['version']}")
    print(f"행 수: {loaded_data['dataset_info']['rows']:,}")
    
    return loaded_data

# JSON 성능 비교
def json_performance_comparison():
    """표준 JSON vs Atio JSON 성능 비교"""
    import time
    import json as std_json
    
    # 테스트 데이터 생성
    test_data = {
        'items': [{'id': i, 'value': f'item_{i}'} for i in range(10000)]
    }
    
    # 표준 JSON 처리
    start_time = time.time()
    with open('standard.json', 'w') as f:
        std_json.dump(test_data, f)
    std_write_time = time.time() - start_time
    
    start_time = time.time()
    with open('standard.json', 'r') as f:
        std_data = std_json.load(f)
    std_read_time = time.time() - start_time
    
    # Atio JSON 처리
    start_time = time.time()
    atio.write_json(test_data, 'atio.json')
    atio_write_time = time.time() - start_time
    
    start_time = time.time()
    atio_data = atio.read_json('atio.json')
    atio_read_time = time.time() - start_time
    
    # 결과 비교
    print("📊 JSON 처리 성능 비교:")
    print(f"표준 JSON 쓰기: {std_write_time:.4f}초")
    print(f"Atio JSON 쓰기: {atio_write_time:.4f}초")
    print(f"표준 JSON 읽기: {std_read_time:.4f}초")
    print(f"Atio JSON 읽기: {atio_read_time:.4f}초")
    
    # 안전성 검증

# 프로세스 풀 사용 예시
def process_pool_example():
    """프로세스 풀을 사용한 병렬 처리"""
    import atio
    from concurrent.futures import ProcessPoolExecutor
    
    def process_chunk(chunk_data):
        """개별 청크 처리 함수"""
        # 데이터 처리 시뮬레이션
        result = chunk_data * 2
        return result
    
    # 프로세스 풀 가져오기
    process_pool = atio.get_process_pool()
    
    # 대용량 데이터를 청크로 분할
    data = list(range(1000))
    chunk_size = 100
    chunks = [data[i:i+chunk_size] for i in range(0, len(data), chunk_size)]
    
    # 병렬 처리
    with ProcessPoolExecutor(max_workers=4) as executor:
        results = list(executor.map(process_chunk, chunks))
    
    print(f"✅ {len(chunks)}개 청크 병렬 처리 완료")
    print(f"처리된 총 항목: {sum(len(chunk) for chunk in chunks)}")
    
    return results
