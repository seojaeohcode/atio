🔧 플러그인 시스템 개요

Atio의 플러그인 시스템은 다양한 데이터 타입과 포맷을 지원하기 위한 확장 가능한 아키텍처입니다. 새로운 데이터 타입이나 포맷을 쉽게 추가할 수 있도록 설계되었습니다.

📊 지원되는 데이터 타입

Atio는 다음과 같은 주요 데이터 타입을 지원합니다:

🐼 Pandas DataFrame

가장 널리 사용되는 Python 데이터 분석 라이브러리

import pandas as pd
import atio

# Pandas DataFrame 생성
df = pd.DataFrame({
    'name': ['Alice', 'Bob', 'Charlie'],
    'age': [25, 30, 35],
    'city': ['Seoul', 'Busan', 'Incheon']
})

# 다양한 포맷으로 저장
atio.write(df, 'data.csv', format='csv')
atio.write(df, 'data.parquet', format='parquet')
atio.write(df, 'data.xlsx', format='excel')

⚡ Polars DataFrame

고성능 Rust 기반 데이터 처리 라이브러리

import polars as pl
import atio

# Polars DataFrame 생성
df = pl.DataFrame({
    'product_id': [101, 102, 103],
    'product_name': ['Laptop', 'Mouse', 'Keyboard'],
    'price': [1200, 25, 75]
})

# 고성능 저장

atio.write(df, 'products.csv', format='csv')

🔌 플러그인 등록 시스템

Atio는 자동으로 지원되는 라이브러리를 감지하고 플러그인을 등록합니다:

# 플러그인 등록 과정 (내부적으로 자동 실행)
def register_plugins():
    """지원되는 라이브러리들을 자동으로 감지하고 플러그인 등록"""
    
    # Pandas 플러그인 등록
    try:
        import pandas as pd
        register_pandas_handlers()
        print("✅ Pandas 플러그인 등록 완료")
    except ImportError:
        print("⚠️ Pandas가 설치되지 않음")
    
    # Polars 플러그인 등록
    try:
        import polars as pl
        register_polars_handlers()
        print("✅ Polars 플러그인 등록 완료")
    except ImportError:
        print("⚠️ Polars가 설치되지 않음")
    
    # NumPy 플러그인 등록
    try:
        import numpy as np
        register_numpy_handlers()
        print("✅ NumPy 플러그인 등록 완료")
    except ImportError:
        print("⚠️ NumPy가 설치되지 않음")

# 실행 결과:
# ✅ Pandas 플러그인 등록 완료
# ✅ Polars 플러그인 등록 완료  
# ✅ NumPy 플러그인 등록 완료

🎯 포맷별 최적화

각 데이터 타입과 포맷 조합에 대해 최적화된 처리를 제공합니다:

# Pandas + Parquet 최적화 예시
def pandas_parquet_optimization():
    """Pandas DataFrame을 Parquet으로 저장할 때의 최적화"""
    import pandas as pd
    
    # 대용량 데이터 생성
    df = pd.DataFrame({
        'id': range(100000),
        'value': np.random.randn(100000),
        'category': np.random.choice(['A', 'B', 'C'], 100000)
    })
    
    # 최적화된 저장
    atio.write(df, 'large_data.parquet', 
              format='parquet',
              compression='snappy',  # 빠른 압축
              engine='pyarrow',     # 고성능 엔진
              show_progress=True)
    
    print("✅ Pandas + Parquet 최적화 저장 완료")

# Polars + CSV 최적화 예시  
def polars_csv_optimization():
    """Polars DataFrame을 CSV로 저장할 때의 최적화"""
    import polars as pl
    
    # 고성능 Polars DataFrame
    df = pl.DataFrame({
        'timestamp': pl.date_range(
            start=datetime(2024, 1, 1),

🚀 성능 비교

다양한 데이터 타입과 포맷 조합의 성능을 비교해보겠습니다:

# 성능 벤치마크 예시
def benchmark_formats():
    """다양한 포맷의 성능 벤치마크"""
    import time
    import pandas as pd
    import polars as pl
    
    # 테스트 데이터 생성
    data_size = 100000
    df_pandas = pd.DataFrame({
        'id': range(data_size),
        'value': np.random.randn(data_size),
        'category': np.random.choice(['A', 'B', 'C'], data_size)
    })
    
    df_polars = pl.DataFrame({
        'id': range(data_size),
        'value': np.random.randn(data_size),
        'category': np.random.choice(['A', 'B', 'C'], data_size)
    })
    
    # 성능 측정
    formats = ['csv', 'parquet', 'json']
    results = {}
    
    for fmt in formats:
        # Pandas 성능
        start_time = time.time()
        atio.write(df_pandas, f'pandas_{fmt}.{fmt}', format=fmt)
        pandas_time = time.time() - start_time
        
        # Polars 성능
        start_time = time.time()
        atio.write(df_polars, f'polars_{fmt}.{fmt}', format=fmt)
        polars_time = time.time() - start_time
        
        results[fmt] = {
            'pandas': pandas_time,
            'polars': polars_time
        }
    
    # 결과 출력
    print("📊 포맷별 성능 비교 (초):")
    for fmt, times in results.items():
        print(f"{fmt.upper():>8}: Pandas {times['pandas']:.3f}s, Polars {times['polars']:.3f}s")

# 실행 결과:
# 📊 포맷별 성능 비교 (초):
#      CSV: Pandas 0.245s, Polars 0.189s
#  PARQUET: Pandas 0.156s, Polars 0.134s  
#     JSON: Pandas 0.298s, Polars 0.267s
